class MathBox {

  choose(a : int, b : int, flag : boolean): int {
    volatile result : int := 0

    if (flag) {
      result := a + b
    } else {
      result := a - b
    }

    return result
  }

  sumTo(n : int): int {
    volatile i : int := 0
    volatile total : int := 0

    for (i := 0, i <= n, i := i + 1) {
      total := total + i
    }

    return total
  }
}

main(): int {
  volatile m : MathBox := MathBox()
  volatile x : int := m.choose(10, 3, true)
  volatile y : int := m.sumTo(5)

  if (x > y) {
    print(x)
  } else {
    print(y)
  }

  volatile i : int := 0
  for (i := 0, i < 3, i := i + 1) {
    if (i = 1) {
      print(i)
    } else {
      print(x)
    }
  }

  return 0
}
